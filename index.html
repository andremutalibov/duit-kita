/** * Duit Kita v2 — expense & income logger backend * Paste into Apps Script attached to your Google Sheet, then redeploy: * Deploy → Manage deployments → Edit (pencil) → Version: New version → Deploy. * The /exec URL stays the same. */ const SHEET_NAME = 'Expenses'; const TZ = 'Asia/Jakarta'; const ALLOWED_NAMES = ['Andre', 'Dinda']; const EXPENSE_CATEGORIES = ['Food', 'Transport', 'Shopping', 'Bills', 'Other']; const INCOME_CATEGORIES = ['Salary', 'Bonus', 'Transfer', 'Other']; const ALLOWED_SOURCES = [ 'Seabank Andre', 'Seabank Dinda', 'BCA Andre', 'BCA Dinda', 'Ewallet Andre', 'Ewallet Dinda' ]; const MAX_DAYS_BACK = 30; const HEADER = ['Date', 'Logged At', 'Name', 'Category', 'Amount', 'Source', 'Type']; function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) sh = ss.insertSheet(SHEET_NAME); if (sh.getLastRow() === 0) { sh.appendRow(HEADER); sh.getRange('1:1').setFontWeight('bold'); sh.setFrozenRows(1); } else { // migrate v1 sheets: add Type column header if missing const firstRow = sh.getRange(1, 1, 1, 7).getValues()[0]; if (String(firstRow[6]) !== 'Type') { sh.getRange(1, 7).setValue('Type').setFontWeight('bold'); } } return sh; } function doPost(e) { try { const data = JSON.parse(e.postData.contents); if (data.action === 'list') { const month = String(data.month || ''); if (!/^\d{4}-\d{2}$/.test(month)) return json_(false, 'Bad month format'); const sh = getSheet_(); const last = sh.getLastRow(); const entries = []; if (last > 1) { const rows = sh.getRange(2, 1, last - 1, 7).getValues(); rows.forEach(function (r) { const dateStr = r[0] instanceof Date ? Utilities.formatDate(r[0], TZ, 'yyyy-MM-dd') : String(r[0]); if (dateStr.indexOf(month) !== 0) return; entries.push({ date: dateStr, name: String(r[2]), category: String(r[3]), amount: Number(r[4]) || 0, source: String(r[5]), type: String(r[6]) === 'Income' ? 'Income' : 'Expense' }); }); } return ContentService .createTextOutput(JSON.stringify({ ok: true, month: month, entries: entries })) .setMimeType(ContentService.MimeType.JSON); } const type = data.type === 'Income' ? 'Income' : 'Expense'; const cats = type === 'Income' ? INCOME_CATEGORIES : EXPENSE_CATEGORIES; if (ALLOWED_NAMES.indexOf(data.name) === -1) return json_(false, 'Unknown name'); if (cats.indexOf(data.category) === -1) return json_(false, 'Unknown category'); if (ALLOWED_SOURCES.indexOf(data.source) === -1) return json_(false, 'Unknown source'); const amount = Number(data.amount); if (!(amount > 0)) return json_(false, 'Invalid amount'); if (!/^\d{4}-\d{2}-\d{2}$/.test(String(data.date))) return json_(false, 'Bad date format'); const d = new Date(data.date + 'T00:00:00'); const today = new Date(); today.setHours(0, 0, 0, 0); const min = new Date(today); min.setDate(min.getDate() - MAX_DAYS_BACK); if (isNaN(d.getTime()) || d > today || d < min) return json_(false, 'Date out of allowed range'); const sh = getSheet_(); sh.appendRow([data.date, new Date(), data.name, data.category, amount, data.source, type]); return json_(true); } catch (err) { return json_(false, String(err)); } } function doGet(e) { try { return json_(true, 'duit-kita backend v2 is alive'); } catch (err) { return json_(false, String(err)); } } function json_(ok, msg) { const body = ok ? { ok: true, message: msg || 'saved' } : { ok: false, error: msg }; return ContentService .createTextOutput(JSON.stringify(body)) .setMimeType(ContentService.MimeType.JSON); }
